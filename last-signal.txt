function chargeTick(s, dt) {
  if (!CHARGE.held)
    return;
  if (s.p.reload || s.p.heal || s.p.swap || s.phase !== 'alive') {
    stopCharge(true);
    return;
  }
  CHARGE.time = Math.min(1.65, CHARGE.time + dt);
  let q = CHARGE.time / 1.65, stage = Math.min(5, Math.ceil(q * 5));
  audio?.gbeChargeMove(eyeOf(s.p), q);
  if (stage > CHARGE.audioLevel) {

    audio?.gbeChargeStage(eyeOf(s.p), stage);
  }
}

function sphereRay(o, d, c, r) {
  let oc = sub(o, c), b = dot(oc, d), disc = b * b - dot(oc, oc) + r * r;
  if (disc < 0)
    return Infinity;
  let t = -b - Math.sqrt(disc);
  return t > 0 ? t : Infinity;
}

function floorAt(x, z, preferredY = null) {
  let best = null;
  for (const f of WORLD.floors) {
    if (x < f.x0 - 0.0001 || x > f.x1 + 0.0001 || z < f.z0 - 0.0001 || z > f.z1 + 0.0001)
      continue;
    let y = f.axis ? mix(f.y0, f.y1, clamp(f.axis === 'x' ? (x - f.x0) / (f.x1 - f.x0) : (z - f.z0) / (f.z1 - f.z0), 0, 1)) : f.y;
    if (preferredY !== null && y > preferredY + 0.34)
      continue;
    if (!best || y > best.y + 0.000001)
      best = {
        ...f,
        y
      };
  }
  return best;
}

function blocked(x, y, z, r = 0.24, head = 1.62) {
  const hit = c => y + head > c.y0 + 0.03 && y + 0.18 < c.y1 && x + r > c.x0 && x - r < c.x1 && z + r > c.z0 && z - r < c.z1;
  if (WORLD.colliders.some(hit))
    return true;
  for (let f of WORLD.floors)
    if (f.axis && x + r > f.x0 && x - r < f.x1 && z + r > f.z0 && z - r < f.z1) {
      let top = mix(f.y0, f.y1, clamp((z - f.z0) / (f.z1 - f.z0), 0, 1));
      if (y < top - 0.34 && y + head > top - 0.22)
        return true;
    }
  for (const d of WORLD.doors)
    if (!doorOpen(d.id) && hit(doorBox(d)))
      return true;
  for (const b of WORLD.breaches)
    if (!S.defense.breaches[b.id] && hit(breachBox(b)))
      return true;
  return false;
}

  for (const c of WORLD.colliders)
    if (!c.playerOnly && c.tag !== ignore)
      near = Math.min(near, rayBox(o, d, c));
  for (const q of WORLD.doors)
    if (q.id !== ignore && !doorOpen(q.id))
      near = Math.min(near, rayBox(o, d, doorBox(q)));
  for (const q of WORLD.breaches)
    if (!S.defense.breaches[q.id])
      near = Math.min(near, rayBox(o, d, breachBox(q)));
  return near;
}
function canSee(a, b) {
  const dv = sub(b, a), distance = len(dv);
  return rayWorld(a, scale(dv, 1 / (distance || 1))) > distance - 0.18;
}
function noise(s, radius, kind, x = s.p.x, z = s.p.z) {
  for (const e of s.enemies) {
    if (e.hp <= 0)
      continue;
    const occluded = !canSee([
      e.x,
      e.y + 1.5,
      e.z
    ], [
      x,
      s.p.y + 1,
      z
    ]);
    if (Math.hypot(e.x - x, e.z - z) < radius * (occluded ? 0.58 : 1)) {
      e.target = {
        x,
        z
      };
      e.alert = Math.max(e.alert, kind === 'gunshot' ? 24 : 7);
      if (e.state !== 'breach')
        e.state = kind === 'gunshot' ? 'hunt' : 'investigate';
      e.pathTimer = 0;
    }
  }
}
function hurt(s, amount, cause) {
  if (s.phase !== 'alive')
    return;
  const p = s.p;
  p.hp = Math.max(0, p.hp - amount);
  p.lastHurt = s.time;
  p.hurt = 1;
  p.shake = Math.min(1, p.shake + 0.55);
  p.adrenaline = 1.4;
  p.bleed = 0;
  p.injury = 0;
  audio?.impact(eyeOf(p));
  if (p.hp <= 0) {
    s.phase = 'dead';
    s.deathReason = cause;
    CHARGE.held = false;
    CHARGE.time = 0;
    input.fire = false;
    s.defense.travel = null;
  }
}
function awardTrace(amount, source = 'contact') {
  const s = S, d = s.defense, base = source === 'kill' ? 60 + (amount >= 20 ? 40 : 0) : source === 'melee' ? 130 : 10, n = base * (d.doubleUntil > s.time ? 2 : 1);
  d.trace = Math.min(999999, d.trace + n);
  s.pointFx.push({
    n,
    at: s.time
  });
  s.pointFx = s.pointFx.slice(-6);
  if (source === 'kill' || source === 'melee') {
    d.kills++;
    if (amount >= 20 && source === 'kill')
      d.headshots++;
    if (d.kills % 8 === 0)
      spawnDrop(s);
  }
}

function spawnEnemy(s) {
  if (s.defense.monkeyRound)
    return spawnMonkey(s);
  const d = s.defense, available = WORLD.breaches.filter(b => d.rooms[b.room]);
  if (!available.length)
    return false;
  const idx = ++d.spawned, b = available[(idx - 1) % available.length], fast = d.wave >= 4 && (idx % 3 === 0 || d.wave >= 8), hp = Math.round(125 * Math.pow(1.115, Math.min(30, d.wave - 1)));
  openBreach(s, b);
  s.enemies.push({
    id: 'Z' + d.wave + '-' + idx,
    archetype: fast ? 'zombie-runner' : 'zombie-walker',
    variant: idx % 3,
    x: b.spawn.x,
    y: floorAt(b.spawn.x, b.spawn.z)?.y || 0,
    z: b.spawn.z,
    hp,
    maxHp: hp,
    state: 'breach',
    breachId: b.id,
    boardClock: 0.72,
    yaw: Math.atan2(b.inside.x - b.spawn.x, b.inside.z - b.spawn.z),
    target: {
      x: s.p.x,
      y: s.p.y,
      z: s.p.z
    },
    path: [],
    pathTimer: 0,
    alert: 35,
    attack: 0,
    attackResolved: false,
    attackDamage: 50,
    anim: s.rng() * 6,
    step: 0,
    deadAt: 0,
    entryFrom: { ...b.spawn },
    entryTo: { ...b.inside },
    ingressT: 0,
    ingressDuration: 1.45,
    stagger: 0,
    knockX: 0,
    knockZ: 0
  });
  return true;
}
function waveStep(s, dt) {
  const d = s.defense;
  d.comboTimer = Math.max(0, d.comboTimer - dt);
  if (!d.comboTimer)
    d.combo = 0;
  d.pressure = Math.max(0, d.pressure - dt * 0.65);
  d.spawnClock -= dt;
  if (d.queue > 0 && d.spawnClock <= 0 && s.enemies.filter(e => e.hp > 0).length < 24 && spawnEnemy(s)) {
    d.queue--;
    d.spawnClock = d.monkeyRound ? 2 : Math.max(0.7, 2.2 - (d.wave - 1) * 0.14);
  }
  if (d.queue === 0 && !s.enemies.some(e => e.hp > 0)) {
    if (d.intermission === 0) {
      d.intermission = 9;
      if (d.monkeyRound) {
        d.pickups.push({
          type: 'ammo',
          x: s.p.x + 0.8,
          y: s.p.y + 0.3,
          z: s.p.z,
          at: s.time,
          life: 18
        });
        say(s, 'SPACE MONKEYS CLEARED \xB7 MAX AMMO', 4);
      } else
        say(s, 'ROUND ' + d.wave + ' SURVIVED', 4);
      audio?.roundCue(false);
    } else {
      d.intermission = Math.max(0, d.intermission - dt);
      if (d.intermission === 0) {
        d.wave++;
        prepareRound(s);
        s.p.grenadeCount = Math.min(4, s.p.grenadeCount + 2);
        audio?.roundCue(true);
        say(s, d.monkeyRound ? 'SPACE MONKEYS \xB7 PROTECT YOUR PERKS' : 'ROUND ' + d.wave, 4);
      }
    }
  }
  s.enemies = s.enemies.filter(e => e.hp > 0 || s.time - e.deadAt < 4);
}
function enemyTick(s, e, dt) {
  if (e.archetype === 'space-monkey')
    return monkeyTick(s, e, dt);
  if (e.hp <= 0)
    return;
  const p = s.p;
  e.stagger = Math.max(0, e.stagger - dt);
  e.knockX *= Math.exp(-dt * 7);
  e.knockZ *= Math.exp(-dt * 7);
  e.attack = Math.max(0, e.attack - dt);
  e.anim += dt * (e.archetype === 'zombie-runner' ? 3 : 1.7);
  if (e.state === 'breach') {
    if (e.ingressT === 0 && s.defense.boards[e.breachId] > 0) {
      e.boardClock -= dt;
      if (e.boardClock <= 0) {
        s.defense.boards[e.breachId]--;
        e.boardClock = 0.72;
        const b = WORLD.breaches.find(b => b.id === e.breachId);
        dirtBurst(s, [
          b.x,
          1.5,
          b.z
        ], 5);
        audio?.boardBreak([
          b.x,
          1.5,
          b.z
        ]);
      }
      return;
    }
    e.ingressT = Math.min(1, e.ingressT + dt / e.ingressDuration);
    let t = e.ingressT * e.ingressT * (3 - 2 * e.ingressT);
    e.x = mix(e.entryFrom.x, e.entryTo.x, t);
    e.z = mix(e.entryFrom.z, e.entryTo.z, t);
    e.y = mix(floorAt(e.entryFrom.x, e.entryFrom.z)?.y || 0, floorAt(e.entryTo.x, e.entryTo.z)?.y || 0, t);
    if (e.ingressT === 1) {
      e.state = 'hunt';
      e.pathTimer = 0;
    }
    return;
  }
  const distance = dist2(e, p), sameLevel = Math.abs(p.y - e.y) < 1.25, eyes = [
      e.x,
      e.y + 1.75,
      e.z
    ], visible = sameLevel && canSee(eyes, eyeOf(p));
  e.target = {
    x: p.x,
    y: p.y,
    z: p.z
  };
  if (visible && distance < 1.55 && e.attack === 0 && e.stagger === 0 && !s.defense.travel) {
    e.attack = 1.04;
    e.attackResolved = false;
    audio?.zombieTell(eyes);
  }
  if (e.attack > 0 && !e.attackResolved && e.attack < 0.27) {
    e.attackResolved = true;
    if (dist2(e, p) < 1.68 && Math.abs(p.y - e.y) < 1.25 && canSee([
        e.x,
        e.y + 1.1,
        e.z
      ], eyeOf(p)) && !s.defense.travel)
      hurt(s, e.attackDamage, 'zombies');
    audio?.zombieStrike(eyes);
  }
  const direct = visible && sameLevel && walkableSegment(e, p);
  e.pathTimer -= dt;
  if (e.pathTimer <= 0) {
    e.path = direct ? [{
        x: p.x,
        y: p.y,
        z: p.z
      }] : pathBetween(e, p);
    e.pathTimer = 0.62;
  }
  while (e.path.length > 1 && dist2(e, e.path[0]) < 0.42)
    e.path.shift();
  const goal = e.path[0];
  if (!goal)
    return;
  let dx = goal.x - e.x, dz = goal.z - e.z, L = Math.hypot(dx, dz) || 1, speed = e.archetype === 'zombie-runner' ? 2.75 : Math.min(2.35, 1.13 + (s.defense.wave - 1) * 0.12);
  if (e.attack > 0.08)
    speed *= 0.22;
  if (e.stagger > 0)
    speed *= 0.14;
  let nx = e.x + dx / L * speed * dt + e.knockX * dt, nz = e.z + dz / L * speed * dt + e.knockZ * dt;
  e.yaw += angleDelta(Math.atan2(dx, dz), e.yaw) * Math.min(1, dt * 4.5);
  let f = floorAt(nx, nz, e.y);
  if (f && Math.abs(f.y - e.y) < 0.48 && !blocked(nx, f.y, nz, 0.25, 1.8)) {
    e.x = nx;
    e.z = nz;
    e.y = f.y;
  } else {
    let a = floorAt(nx, e.z, e.y);
    if (a && !blocked(nx, a.y, e.z, 0.25, 1.8)) {
      e.x = nx;
      e.y = a.y;
    } else {
      a = floorAt(e.x, nz, e.y);
      if (a && !blocked(e.x, a.y, nz, 0.25, 1.8)) {
        e.z = nz;
        e.y = a.y;
      } else
        e.pathTimer = 0;
    }
  }
  e.step += speed * dt;
  if (e.step > 1.3) {
    e.step = 0;
    if (distance < 20)
      audio?.zombieFootstep(floorSurface(floorAt(e.x, e.z, e.y)), 0.54, [
        e.x,
        e.y + 0.1,
        e.z
      ]);
  }
}
function interactable(s) {
  const p = s.p, d = s.defense, eye = eyeOf(p), f = forward(p), near = (point, r, ignore) => {
      let v = sub(point, eye), L = len(v);
      return L < r && dot(norm(v), f) > 0.12 && rayWorld(eye, norm(v), ignore) > L - 0.2;
    };
  if (d.travel)
    return null;
  for (let q of WORLD.doors)
    if (!doorOpen(q.id) && near([
        q.x,
        1.65,
        q.z
      ], 3.1, q.id))
      return {
        kind: q.automatic ? 'locked' : 'door',
        id: q.id,
        text: q.automatic ? 'ROCKET SEALED \u2014 LAUNCH TO OPEN' : '[E] Open ' + q.name + ' \xB7 ' + q.cost + ' points'
      };
  for (let m of WORLD.machines)
    if (d.rooms[m.room] && near([
        m.x,
        m.y,
        m.z
      ], 2.7, m.id)) {

      if (m.kind === 'power')

        text = d.perks[m.perk] ? m.name + ' \xB7 ACTIVE' : !d.power ? 'POWER REQUIRED' : '[E] ' + m.name + ' \xB7 ' + m.price + ' points';
      if (m.kind === 'box')
        text = d.boxReady > s.time ? 'CRATE RESETTING\u2026' : '[E] Mystery crate \xB7 950 points';
      if (m.kind === 'launch') {
        let count = [
          'yard',
          'bunker',
          'control'
        ].filter(k => d.landerUsed[k]).length;
        text = d.rocketLaunched ? 'ROCKET LAUNCHED' : !d.power ? 'POWER REQUIRED' : count < 3 ? 'LANDER ROUTES ' + count + '/3 \u2014 RIDE BACK FROM EACH REMOTE PAD' : '[E] Launch rocket';
      }
      if (m.kind === 'upgrade')
        text = !d.rocketLaunched ? 'LAUNCH THE ROCKET FIRST' : (d.weaponTiers[p.weapon] || 0) >= 2 ? 'ARSENAL \xB7 MAXIMUM' : '[E] Upgrade ' + weaponSpec(p.weapon).short + ' \xB7 ' + (5000 + (d.weaponTiers[p.weapon] || 0) * 2500) + ' points';
      return {
        kind: m.kind,

      };
    }
  for (let q of WORLD.gunRacks)
    if (near([
        q.x,
        q.y,
        q.z
      ], 2.7, q.id)) {
      const w = weaponSpec(q.weapon), owned = !!p.ownedWeapons[w.id];
      return {
        kind: 'weapon',
        id: q.id,
        weapon: w.id,
        text: '[E] ' + (owned ? 'Refill ' + w.short + ' \xB7 ' + Math.round(q.price * 0.5) + ' points' : 'Buy ' + w.short + ' \xB7 ' + q.price + ' points')
      };
    }
  for (let q of WORLD.pads)
    if (d.rooms[q.room] && near([
        q.x,
        0.4,
        q.z
      ], 3.1, null)) {
      let text = !d.power ? 'LANDER \xB7 POWER REQUIRED' : q.id === 'home' ? '[E] Lander to courtyard \xB7 250 points' : '[E] Lander to centrifuge \xB7 250 points';
      return {
        kind: 'lander',
        id: q.id,
        text
      };
    }
  const w = nearWindow(s);
  if (w)
    return {
      kind: 'repair',
      id: w.id,
      text: '[F] Rebuild barrier \xB7 10 points / board'
    };
  return null;
}
function interact(s) {
  const target = interactable(s);
  if (!target || s.phase !== 'alive')
    return {
      ok: false,
      error: 'Look at a nearby purchase point.'
    };
  const p = s.p, d = s.defense, spend = n => {
      if (d.trace < n) {
        say(s, 'Need ' + (n - d.trace) + ' more points', 2);
        return false;
      }
      d.trace -= n;
      return true;
    };
  if (target.kind === 'locked')
    return {
      ok: false,
      error: 'Use the landers, then launch the rocket.'
    };
  if (target.kind === 'door') {
    const q = WORLD.doors.find(q => q.id === target.id);
    if (!spend(q.cost))
      return { ok: false };
    d.doors[q.id] = true;
    d.rooms[q.from] = d.rooms[q.to] = true;
    renderer && (renderer.shadowDirty = true);
    audio?.doorOpen([
      q.x,
      1.5,
      q.z
    ]);
    say(s, q.name + ' OPEN', 3);
    return {
      ok: true,
      door: q.id
    };
  }
  if (target.kind === 'weapon') {
    const q = WORLD.gunRacks.find(q => q.id === target.id), w = weaponSpec(q.weapon), owned = !!p.ownedWeapons[w.id];
    if (!spend(owned ? Math.round(q.price * 0.5) : q.price))
      return { ok: false };
    recordAmmo(p);
    p.ownedWeapons[w.id] = {
      mag: w.capacity,
      reserve: w.initialReserve
    };
    if (p.weapon === w.id) {
      p.mag = w.capacity;
      p.reserve = w.initialReserve;
      p.reload = 0;
    } else
      equipWeapon(w.id);
    audio?.weaponPurchase(eyeOf(p), w.id);
    return {
      ok: true,
      weapon: w.id,
      refill: owned
    };
  }
  const m = WORLD.machines.find(m => m.id === target.id);
  if (target.kind === 'power') {
    if (d.power)
      return {
        ok: false,
        error: 'Power is already on.'
      };
    d.power = true;
    d.powerAt = s.time;
    renderer && (renderer.shadowDirty = true);
    audio?.powerOn();
    say(s, 'POWER RESTORED \u2014 LANDERS AND PERK MACHINES ONLINE', 5);
    return {
      ok: true,
      power: true
    };
  }
  if (target.kind === 'perk') {
    if (!d.power)
      return {
        ok: false,
        error: 'Turn on the power first.'
      };
    if (d.perks[m.perk])
      return {
        ok: false,
        error: 'Perk already active.'
      };
    if (!spend(m.price))
      return { ok: false };
    d.perks[m.perk] = true;
    if (m.perk === 'jug') {
      p.maxHp = 200;
      p.hp = Math.min(200, p.hp + 100);
    }
    audio?.perkCue();
    say(s, m.name + ' ACQUIRED', 3);
    return {
      ok: true,
      perk: m.perk
    };
  }
  if (target.kind === 'box') {
    if (d.boxReady > s.time)
      return {
        ok: false,
        error: 'The crate is resetting.'
      };
    if (!spend(950))
      return { ok: false };
    let pool = [
        'morrow',
        'bellwether',
        'vesper',
        'gbe'
      ], id = pool[Math.floor(s.rng() * pool.length)], w = weaponSpec(id);
    recordAmmo(p);
    p.ownedWeapons[id] = {
      mag: w.capacity,
      reserve: w.initialReserve
    };
    if (p.weapon === id) {
      p.mag = w.capacity;
      p.reserve = w.initialReserve;
      p.reload = 0;
    } else
      equipWeapon(id);
    d.boxReady = s.time + 4;
    audio?.perkCue();
    say(s, 'MYSTERY CRATE: ' + w.name.toUpperCase(), 4);
    return {
      ok: true,
      weapon: id
    };
  }
  if (target.kind === 'lander') {
    if (!d.power)
      return {
        ok: false,
        error: 'Turn on the power first.'
      };
    let q = WORLD.pads.find(p => p.id === target.id), destination = q.id === 'home' ? WORLD.pads.find(p => p.id === 'yard') : WORLD.pads.find(p => p.id === 'home');
    if (!d.rooms[destination.room])
      return {
        ok: false,
        error: 'Open the courtyard door first.'
      };
    if (!spend(250))
      return { ok: false };
    stopCharge(true);
    input.fire = false;
    p.reload = p.heal = p.swap = 0;
    d.travel = {
      from: {
        x: q.x,
        y: q.y,
        z: q.z
      },
      to: {
        x: destination.x,
        y: destination.y,
        z: destination.z
      },
      pad: q.id,
      t: 0,
      duration: 4.8
    };
    p.vx = p.vz = p.vy = 0;
    audio?.landerCue();
    say(s, 'LANDER IN TRANSIT', 3);
    return {
      ok: true,
      pad: q.id
    };
  }
  if (target.kind === 'launch') {
    if (d.rocketLaunched)
      return {
        ok: false,
        error: 'Rocket has already launched.'
      };
    if (!d.power || ![
        'yard',
        'bunker',
        'control'
      ].every(k => d.landerUsed[k]))
      return {
        ok: false,
        error: 'Ride the courtyard, bunker and control landers back to the centrifuge.'
      };
    d.rocketLaunched = true;
    d.rocketAt = s.time;
    audio?.rocketCue();
    say(s, 'ROCKET LAUNCHING \u2014 BLAST DOOR OPENING', 5);
    return {
      ok: true,
      launch: true
    };
  }
  if (target.kind === 'upgrade') {
    if (!d.rocketLaunched || !d.doors.rocketDoor)
      return {
        ok: false,
        error: 'Wait for the rocket to clear the launch pad.'
      };
    let t = d.weaponTiers[p.weapon] || 0;
    if (t >= 2)
      return {
        ok: false,
        error: 'Maximum upgrade reached.'
      };
    if (!spend(5000 + t * 2500))
      return { ok: false };
    d.weaponTiers[p.weapon] = ++t;
    d.weaponTier = Math.max(d.weaponTier, t);
    p.mag = weaponSpec(p.weapon).capacity;
    p.reserve = weaponSpec(p.weapon).initialReserve;
    recordAmmo(p);
    audio?.perkCue();
    say(s, weaponSpec(p.weapon).short + ' UPGRADED', 4);
    return {
      ok: true,
      tier: t
    };
  }
  if (target.kind === 'repair')
    return repairBarrier(s);
  return { ok: false };
}

function buildPartyWorld(seed = 417) {
  const g = new Mesh(), glass = new Mesh(), colliders = [], floors = [], signs = [], lights = [], doors = [], breaches = [], gunRacks = [], machines = [], pads = [], nodes = [], groups = [], r = mulberry32(seed);
  const concrete = [
      0.86,
      0.87,
      0.82
    ], brick = [
      0.94,
      0.85,
      0.77
    ], green = [
      0.61,
      0.72,
      0.56
    ], rust = [
      0.78,
      0.61,
      0.44
    ], yellow = [
      0.94,
      0.83,
      0.37
    ];
  function box(x, y, z, w, h, d, mat = 0, t = concrete, solid = true, tag = 'structure') {
    g.box(x, y, z, w, h, d, mat, t);
    if (solid)
      colliders.push({
        x0: x - w / 2,
        x1: x + w / 2,
        y0: y - h / 2,
        y1: y + h / 2,
        z0: z - d / 2,
        z1: z + d / 2,
        tag
      });
  }
  function floor(x0, x1, z0, z1, y, zone, mat = 0) {
    floors.push({
      x0,
      x1,
      z0,
      z1,
      y,
      zone,
      mat
    });

function rayBox(o, d, c) {

      c.y1,

    ];

    if (Math.abs(d[i]) < 1e-7) {
      if (o[i] < min[i] || o[i] > max[i])
        return Infinity;
      continue;
    }
    let a = (min[i] - o[i]) / d[i], b = (max[i] - o[i]) / d[i];
    if (a > b)
      [a, b] = [
        b,
        a
      ];
    lo = Math.max(lo, a);
    hi = Math.min(hi, b);
    if (hi < lo)
      return Infinity;
  }

}
function pathBetween(a, b) {
  const nodes = WORLD.nodes;
  const nearest = p => nodes.reduce((v, n) => Math.hypot(n.x - p.x, n.z - p.z, (n.y - (p.y || 0)) * 3) < Math.hypot(v.x - p.x, v.z - p.z, (v.y - (p.y || 0)) * 3) ? n : v, nodes[0]).i;
  const start = nearest(a), end = nearest(b), open = new Set([start]), g = nodes.map(() => Infinity), prev = nodes.map(() => -1), done = new Set();
  g[start] = 0;
  let reached = false;
  while (open.size) {
    let cur = -1, best = Infinity;
    for (let i of open) {
      const f = g[i] + dist2(nodes[i], nodes[end]);
      if (f < best) {
        cur = i;
        best = f;
      }
    }
    if (cur === end) {
      reached = true;
      break;
    }
    open.delete(cur);
    done.add(cur);
    for (let n of nodes[cur].links) {
      if (done.has(n) || gateBlocksSegment(nodes[cur], nodes[n]))
        continue;
      let c = g[cur] + dist2(nodes[cur], nodes[n]);
      if (c < g[n]) {
        g[n] = c;
        prev[n] = cur;
        open.add(n);
      }
    }
  }
  if (!reached)
    return [];
  let out = [], cur = end;
  for (let k = 0; k < nodes.length && cur !== -1; k++) {
    out.push({
      x: nodes[cur].x,
      y: nodes[cur].y,
      z: nodes[cur].z
    });
    if (cur === start)
      break;
    cur = prev[cur];
  }
  out.reverse();
  if (out.length > 1 && walkableSegment(a, out[1]))
    out.shift();
  out.push({
    x: b.x,
    y: b.y || 0,
    z: b.z
  });
  return out;
}

function gateBlocksSegment(a, b) {

    return false;

      if (Math.abs(dx) < 0.00001)
        continue;

        continue;

        return true;

      let dz = b.z - a.z;
      if (Math.abs(dz) < 0.00001)
        continue;

      if (Math.abs(x - d.x) < (d.width || 4.65) / 2)

    }
  }
  return false;
}

          vz: (s.rng() - 0.5) * 3,
          life: 0.22 + s.rng() * 0.22,
          mat: 10
        });
    }
    if (audio) {
      audio.gbeShot(o, charge);
      for (let target of targets.slice(0, 3)) {
        let pos = [
          target.e.x,
          target.e.y + 1.35,
          target.e.z
        ];
        if (target.weak)
          audio.critical(pos);
        else
          audio.zombieHit(pos, charge);
      }
      if (!targets.length)
        audio.beamImpact(add(eyeOf(p), scale(d, beamRange)), charge);
    }
  }
  return {
    ok: true,
    hit: hit?.e.id || null,
    hits: hitRecords.map(v => v.id),
    hitRecords,
    pierces,
    optic: !!(hit && hit.weak),
    critical,
    damage: totalDamage,
    charge,
    chargeLevel: beam.chargeLevel,
    beam: {
      id,
      start: o,
      end,
      life,
      range: beamRange,
      maxRange,
      pierces,
      pierce: targets.length,
      hitIds: beam.hitIds.slice(),
      impactPoints: impactPoints.map(v => v.slice())
    }
  };
}
function gunShot(s) {
  const p = s.p;
  if (s.phase !== 'alive')
    return { ok: false };
  let owner = 'local';
  ensureLoadout(p);
  let spec = {
    ...weaponSpec(p.weapon),
    ...gunStats(s)
  };
  if (spec.id === 'gbe')
    return {
      ok: false,
      error: 'Hold and release the Arc Projector.'
    };
  if (!p || false || p.mag <= 0 || p.reload > 0 || p.heal > 0 || p.swap > 0 || (p.shotCd || 0) > 0.00001)
    return {
      ok: false,
      error: p?.mag <= 0 ? 'empty magazine' : 'weapon unavailable'
    };
  let id = owner + ':' + ++S.shotSeq;
  p.mag--;
  recordAmmo(p);
  p.shotCd = spec.cooldown;
  p.flash = 0.105;
  let yaw = Number(p.yaw) || 0, pitch = clamp((Number(p.pitch) || 0) + (p.recoil || 0), -1.35, 1.3), base = [
      Math.sin(yaw) * Math.cos(pitch),
      Math.sin(pitch),
      Math.cos(yaw) * Math.cos(pitch)
    ], right = [
      Math.cos(yaw),
      0,
      -Math.sin(yaw)
    ], up = norm(cross(right, base)), eye = [
      p.x,
      p.y + (p.height || 1.62) - 0.16,
      p.z
    ], o = [
      eye[0] + right[0] * 0.32 + base[0] * 0.48,
      eye[1],
      eye[2] + right[2] * 0.32 + base[2] * 0.48
    ], seed = 2166136261;
  for (let ch of String(id)) {
    seed ^= ch.charCodeAt(0);
    seed = Math.imul(seed, 16777619);
  }
  let random = mulberry32(seed >>> 0), count = spec.mode === 'scatter' ? spec.pellets : 1, rays = [], hits = new Map(), aimed = p.aim > 0.7, spread = aimed ? spec.aimedSpread : spec.hipSpread, range = spec.range;
  for (let i = 0; i < count; i++) {
    let h = (random() * 2 - 1) * spread, v = (random() * 2 - 1) * spread * 0.72, dir = norm(add(base, add(scale(right, h), scale(up, v)))), wall = rayWorld(eyeOf(p), dir), endDist = Math.min(range, wall), best = null, weak = false;
    for (let e of s.enemies) {
      if (e.hp <= 0)
        continue;
      let head = sphereRay(eyeOf(p), dir, [
          e.x,
          e.y + (e.headHeight || 1.82),
          e.z
        ], e.headRadius || 0.34), body = sphereRay(eyeOf(p), dir, [
          e.x,
          e.y + (e.bodyHeight || 1.13),
          e.z
        ], e.bodyRadius || 0.46), t = Math.min(head, body);
      if (t < endDist - 0.02 && t > 0) {
        endDist = t;
        best = e;
        weak = head < body;
      }
    }
    let end = add(eyeOf(p), scale(dir, endDist));
    rays.push({
      start: o.slice(),
      end,
      hit: !!best,
      enemy: best?.id || null,
      optic: weak
    });
    if (best) {
      let falloff = 1 - Math.min(0.52, endDist / range * 0.52), amount = Math.max(1, Math.round((weak ? spec.opticDamage : spec.bodyDamage) * falloff)), rec = hits.get(best.id) || {
          e: best,
          damage: 0,
          optic: false,
          distance: endDist,
          pellets: 0
        };
      rec.damage += amount;
      rec.optic = rec.optic || weak;
      rec.distance = Math.min(rec.distance, endDist);
      rec.pellets++;
      hits.set(best.id, rec);
    }
  }
  let hitRecords = [], totalDamage = 0;
  for (let rec of hits.values()) {
    let dealt = rec.damage;
    {
      let before = rec.e.hp;
      rec.e.hp = Math.max(0, rec.e.hp - (s.defense.instaUntil > s.time ? Math.max(rec.e.hp, rec.damage) : rec.damage));
      dealt = before - rec.e.hp;
      rec.e.target = {
        x: p.x,
        z: p.z
      };
      rec.e.alert = 32;
      rec.e.pathTimer = 0;
      rec.e.stagger = Math.min(1.1, (rec.e.stagger || 0) + 0.22 + rec.damage / 180);
      rec.e.knockX = (rec.e.knockX || 0) + base[0] * 1.1;
      rec.e.knockZ = (rec.e.knockZ || 0) + base[2] * 1.1;
      rec.e.state = rec.e.hp <= 0 ? 'inert' : rec.e.state === 'breach' ? 'breach' : 'hunt';
      if (rec.e.hp <= 0)
        rec.e.deadAt = s.time;
      awardTrace(rec.optic ? 22 : 12, rec.e.hp <= 0 ? 'kill' : 'contact', owner);
    }
    totalDamage += dealt;
    hitRecords.push({
      id: rec.e.id,
      optic: rec.optic,
      damage: dealt,
      killed: rec.e.hp <= 0,
      distance: Number(rec.distance.toFixed(2)),
      pellets: rec.pellets
    });
  }
  let beamRange = Math.min(range, rayWorld(eyeOf(p), base)), end = add(eyeOf(p), scale(base, beamRange)), hit = hitRecords[0] || null, impactPoints = [...hits.values()].map(v => [
      v.e.x,
      v.e.y + 1.35,
      v.e.z
    ]), life = spec.mode === 'scatter' ? 0.19 : 0.15, beam = {
      id,
      weapon: spec.id,
      charge: 0,
      chargeLevel: 0,
      start: o,
      end,
      at: s.time,
      life,
      hit: hitRecords.length > 0,
      critical: hitRecords.some(v => v.optic),
      range: beamRange,
      maxRange: range,
      pierces: 0,
      pierce: hitRecords.length,
      hitIds: hitRecords.map(v => v.id),
      impactPoints,
      rays,
      kind: spec.mode === 'scatter' ? 'eight-pellet spread' : 'ballistic tracer'
    };
  if (!Array.isArray(s.beams))
    s.beams = [];
  s.beams.push(beam);
  s.beams = s.beams.slice(-24);
  let sh = S.defense, now = s.time;
  if (hitRecords.length) {
    sh.combo = now - (sh.lastDischargeAt || -99) < 1.35 ? Math.min(8, (sh.combo || 0) + 1) : 1;
    sh.comboTimer = 1.35;
    sh.lastDischargeAt = now;
  }
  s.feedback = {
    at: now,
    hit: hitRecords.length > 0,
    critical: beam.critical,
    damage: totalDamage,
    enemy: hit?.id || null,
    hits: hitRecords.map(v => v.id),
    pierces: 0,
    shotId: id,
    chargeLevel: 0,
    weapon: spec.id
  };
  let recoil = spec.recoil;
  p.recoil = (p.recoil || 0) + recoil;
  p.shake = Math.min(1, (p.shake || 0) + recoil + (spec.mode === 'automatic' ? 0.035 : 0.1));
  p.vx = (p.vx || 0) - base[0] * (spec.mode === 'scatter' ? 0.55 : 0.18);
  p.vz = (p.vz || 0) - base[2] * (spec.mode === 'scatter' ? 0.55 : 0.18);
  noise(s, spec.mode === 'scatter' ? 74 : 56, 'gunshot', p.x, p.z, p);
  {
    for (let r of rays) {
      let pos = r.end, motes = spec.mode === 'scatter' ? 2 : 5;
      for (let j = 0; j < motes; j++)
        s.particles.push({
          x: pos[0] + (random() - 0.5) * 0.18,
          y: pos[1] + (random() - 0.5) * 0.18,
          z: pos[2] + (random() - 0.5) * 0.18,
          vx: (random() - 0.5) * 2.8,
          vy: 0.3 + random() * 1.8,
          vz: (random() - 0.5) * 2.8,
          life: 0.12 + random() * 0.18,
          mat: r.hit ? 10 : 9
        });
    }
    audio?.weaponShot(spec.id, o, hitRecords.length);
  }
  return {
    ok: true,
    weapon: spec.id,
    hit: hit?.id || null,
    hits: hitRecords.map(v => v.id),
    hitRecords,
    pellets: count,
    critical: beam.critical,
    damage: totalDamage,
    beam: {
      id,
      start: o,
      end,
      life,
      range: beamRange,
      maxRange: range,
      rays: rays.map(v => ({
        start: v.start.slice(),
        end: v.end.slice(),
        hit: v.hit,
        enemy: v.enemy,
        optic: v.optic
      }))
    }
  };
}
function reload(s) {
  if (s.phase !== 'alive')
    return { ok: false };
  stopCharge(true);
  let p = s.p;
  if (p.heal || p.swap || s.phase !== 'alive')
    return {
      ok: false,
      error: 'hands occupied'
    };
  if (p.reload > 0) {
    if (p.reload <= 0.72 && p.mag < p.capacity && p.reserve > 0) {
      p.reload = 0.08;
      p.reloadPerfect = true;
      p.shake = Math.min(1, (p.shake || 0) + 0.18);
      say(s, 'QUICK RELOAD', 1.6);
      audio?.reloadPerfect([
        p.x,
        p.y + 1,
        p.z
      ]);
      return {
        ok: true,
        perfect: true
      };
    }
    return {
      ok: false,
      error: 'reload in progress',
      perfectWindow: 0.72
    };
  }
  if (p.mag === p.capacity || p.reserve <= 0) {
    say(s, p.reserve <= 0 ? 'No loose rounds remain.' : 'The magazine is full.', 3);
    return {
      ok: false,
      error: p.reserve <= 0 ? 'no reserve' : 'magazine full'
    };
  }
  p.reload = reloadSeconds(p);
  p.reloadPerfect = false;
  noise(s, 4, 'magazine');
  audio?.reloadStart([
    p.x,
    p.y + 1,
    p.z
  ], p.weapon);
  return { ok: true };
}
function bandage(s) {
  if (s.phase !== 'alive')
    return { ok: false };
  stopCharge(true);
  let p = s.p;
  if (p.bandages < 1 || p.heal || p.reload || p.swap)
    return {
      ok: false,
      error: 'no dressing or hands occupied'
    };
  if (p.hp >= p.maxHp && p.bleed === 0)
    return {
      ok: false,
      error: 'no injury'
    };
  p.heal = 3.8;
  p.bandages--;
  say(s, 'Pressure. Breathe. Keep the dressing tight.', 4);
  audio?.dressingStart([
    p.x,
    p.y + 1,
    p.z
  ]);
  return { ok: true };
}
function replaceCell(s) {
  if (s.phase !== 'alive')
    return { ok: false };
  stopCharge(true);
  let p = s.p;
  if (p.cells < 1 || p.swap || p.heal || p.reload)
    return {
      ok: false,
      error: 'no spare cell or hands occupied'
    };
  if (p.battery > 115)
    return {
      ok: false,
      error: 'cell still charged'
    };
  p.swap = 1.8;
  p.cells--;
  p.light = false;
  say(s, 'Replacing the cell.', 2);
  audio?.cellSwap([
    p.x,
    p.y + 1,
    p.z
  ]);
  return { ok: true };
}
function tick(s, dt, controls = input) {
  if (s.phase !== 'alive')
    return;
  s.time += dt;
  s.tick++;
  let p = s.p;
  chargeTick(s, dt);
  p.targetPitch = clamp(p.targetPitch, -1.35, 1.3);
  p.yaw += angleDelta(p.targetYaw, p.yaw) * (1 - Math.exp(-dt * 18));
  p.pitch = mix(p.pitch, p.targetPitch, 1 - Math.exp(-dt * 16));
  let f = clamp(controls.forward || 0, -1, 1), strafe = clamp(controls.strafe || 0, -1, 1), amount = Math.hypot(f, strafe);
  if (amount > 1) {
    f /= amount;
    strafe /= amount;
  }
  let moving = amount > 0.02, sliding = false;
  {
    p.slideCooldown = Math.max(0, (p.slideCooldown || 0) - dt);
    p.slideTime = Math.max(0, (p.slideTime || 0) - dt);
    if (controls.slide && controls.sprint && moving && p.slideCooldown <= 0 && p.stamina >= 18 && !p.heal && !p.reload && !p.swap) {
      p.slideTime = 0.78;
      p.slideCooldown = 0.92;
      p.stamina -= 18;
      p.vx *= 1.17;
      p.vz *= 1.17;
    }
    sliding = p.slideTime > 0;
    p.sliding = sliding;
  }
  p.crouch = !!controls.crouch || sliding;
  p.height = mix(p.height, p.crouch ? 1.06 : 1.62, 1 - Math.exp(-dt * 9));
  p.aim = mix(p.aim, controls.aim && !p.reload && !p.heal && !p.swap ? 1 : 0, 1 - Math.exp(-dt * 8));
  p.shotCd = Math.max(0, p.shotCd - dt);
  if (p.shotCd < 0.00001)
    p.shotCd = 0;
  if (s === S && input.fire && weaponSpec(p.weapon).mode === 'automatic' && p.shotCd <= 0.00001)
    fire(s);
  p.recoil *= Math.exp(-dt * 12);
  p.hurt *= Math.exp(-dt * 2.8);
  p.shake = Math.max(0, (p.shake || 0) * Math.exp(-dt * 9));
  p.adrenaline = Math.max(0, (p.adrenaline || 0) - dt);
  p.flash = Math.max(0, p.flash - dt);
  for (let key of [
      'reload',
      'heal',
      'swap'
    ])
    if (p[key] > 0) {
      p[key] = Math.max(0, p[key] - dt);
      if (p[key] === 0) {
        if (key === 'reload') {
          let perfect = !!p.reloadPerfect, n = Math.min(p.capacity - p.mag, p.reserve);
          p.mag += n;
          p.reserve -= n;
          p.reloadPerfect = false;
          recordAmmo(p);
          if (perfect) {
            p.shake = Math.min(1, (p.shake || 0) + 0.16);
            audio?.reloadComplete([
              p.x,
              p.y + 1,
              p.z
            ], true);
          } else
            audio?.reloadComplete([
              p.x,
              p.y + 1,
              p.z
            ], false);
        }
        if (key === 'heal') {
          p.hp = Math.min(p.maxHp, p.hp + 45);
          p.bleed = 0;
          p.injury = Math.max(0, p.injury - 1);
          say(s, 'MEDKIT APPLIED', 2);
          audio?.dressingComplete([
            p.x,
            p.y + 1,
            p.z
          ]);
        }
        if (key === 'swap') {
          p.battery = 120;
          p.light = true;
          audio?.cellSwap([
            p.x,
            p.y + 1,
            p.z
          ], true);
        }
      }
    }
  if (p.stamina < 6)
    p.windLocked = true;
  if (p.stamina > 35)
    p.windLocked = false;
  let sprinting = controls.sprint && moving && !p.windLocked && p.stamina > 5 && !p.crouch && p.aim < 0.4 && p.heal === 0;
  p.sprintFov = mix(p.sprintFov || 0, sprinting || sliding ? 1 : 0, 1 - Math.exp(-dt * 4.4));
  if (sprinting) {
    p.stamina = Math.max(0, p.stamina - dt * PACE.sprintDrain * (s.defense.perks.stamina ? 0.6 : 1) * (p.adrenaline > 0 ? 1.16 : 1));
    p.exert = 1.5;
  } else {
    p.exert = Math.max(0, p.exert - dt);
    if (p.exert <= 0 && !sliding)
      p.stamina = Math.min(100, p.stamina + dt * (p.injury ? 8 : PACE.recovery));
  }
  let speed = p.heal || p.swap || false ? 0.45 : p.reload ? 1.1 : sliding ? PACE.sprint * 1.13 : p.crouch ? PACE.crouch : sprinting ? PACE.sprint : PACE.walk;
  speed *= p.hp < 28 ? 0.68 : p.injury > 1 ? 0.9 : 1;
  if (p.adrenaline > 0)
    speed *= 1.18;
  if (s.defense.perks.stamina && sprinting)
    speed *= 1.15;
  speed *= mix(1, 0.62, p.aim);
  let tx = (Math.sin(p.yaw) * f - Math.cos(p.yaw) * strafe) * speed, tz = (Math.cos(p.yaw) * f + Math.sin(p.yaw) * strafe) * speed, weight = sliding ? moving ? 2 : 1.5 : moving ? PACE.acceleration : PACE.braking;
  p.vx = mix(p.vx, tx, 1 - Math.exp(-dt * weight));
  p.vz = mix(p.vz, tz, 1 - Math.exp(-dt * weight));
  let oldX = p.x, oldZ = p.z, nx = p.x + p.vx * dt, nz = p.z + p.vz * dt;
  if (!blocked(nx, p.y, p.z, 0.24, p.height)) {
    p.x = nx;
  } else
    p.vx = 0;
  if (!blocked(p.x, p.y, nz, 0.24, p.height)) {
    p.z = nz;
  } else
    p.vz = 0;
  let ground = floorAt(p.x, p.z, p.y);
  p.vy -= dt * 14;
  if (ground && p.vy <= 0 && (p.ground && Math.abs(p.y - ground.y) < 0.28 || p.y >= ground.y - 0.28 && p.y + p.vy * dt <= ground.y + 0.08)) {
    let impact = -p.vy;
    p.y = ground.y;
    p.vy = 0;
    if (!p.ground && impact > 7)
      hurt(s, (impact - 6) * 7, 'fall');
    p.ground = true;
  } else {
    p.y += p.vy * dt;
    p.ground = false;
  }
  if (p.y < (WORLD.killY || -160))
    hurt(s, 100, 'fall');
  let displacement = Math.hypot(p.x - oldX, p.z - oldZ);
  p.foot += displacement;
  p.viewBob = Math.sin(p.foot * 8.2) * 0.017 * settings.motion * (moving ? 1 : 0) + Math.sin(s.time * 1.5) * 0.003 * settings.motion;
  p.roll = mix(p.roll, strafe * -0.014 * settings.motion + (moving ? Math.sin(p.foot * 4.1) * 0.008 * settings.motion : 0), 1 - Math.exp(-dt * 5));
  audio?.listen(s);
  let stride = p.crouch ? 0.92 : sprinting ? 1.42 : 1.18;
  if (displacement > 0.0001 && p.ground && Math.floor(p.foot / stride) !== Math.floor((p.foot - displacement) / stride)) {
    let radius = sliding ? 29 : p.crouch ? 2 : sprinting ? 24 : 11;
    if (p.adrenaline > 0)
      radius *= 1.28;
    noise(s, radius, 'footstep');
    audio?.footstep(floorSurface(ground), sliding ? 0.95 : sprinting ? 1 : p.crouch ? 0.25 : 0.6, [
      p.x + Math.cos(p.yaw) * (Math.floor(p.foot / stride) % 2 ? 0.13 : -0.13),
      p.y + 0.055,
      p.z - Math.sin(p.yaw) * (Math.floor(p.foot / stride) % 2 ? 0.13 : -0.13)
    ]);
  }
  if (p.light) {
    p.battery = Math.max(0, p.battery - dt);
    if (p.battery < 0.000001)
      p.battery = 0;
    if (p.battery === 0) {
      p.light = false;
      say(s, 'The cell is exhausted. Replace it in cover. [B]', 6);
    }
  }
  if (s.phase !== 'alive')
    return;
  if (p.bleed > 0) {
    p.hp = Math.max(0, p.hp - p.bleed * dt);
    if (p.hp <= 0)
      hurt(s, 1, 'blood loss');
  }
  if (s.phase !== 'alive')
    return;
  defenseStep(s, dt);
  if (s.phase !== 'alive')
    return;
  waveStep(s, dt);
  for (const e of s.enemies)
    enemyTick(s, e, dt);
  for (const a of s.particles) {
    a.life -= dt;
    a.x += a.vx * dt;
    a.y += a.vy * dt;
    a.z += a.vz * dt;
    a.vy -= dt * 7;
  }
  s.particles = s.particles.filter(a => a.life > 0);
  s.beams = s.beams.filter(b => s.time - b.at < (b.life || 0.28));
  s.captions = s.captions.filter(c => c.end > s.time);
}

  programOf(vs, fs) {
    const gl = this.gl, compile = (type, src) => {
        let sh = gl.createShader(type);
        gl.shaderSource(sh, src);
        gl.compileShader(sh);
        if (!gl.getShaderParameter(sh, gl.COMPILE_STATUS))
          throw new Error(gl.getShaderInfoLog(sh));
        return sh;
      };
    let p = gl.createProgram();
    gl.attachShader(p, compile(gl.VERTEX_SHADER, vs));
    gl.attachShader(p, compile(gl.FRAGMENT_SHADER, fs));
    gl.linkProgram(p);
    if (!gl.getProgramParameter(p, gl.LINK_STATUS))
      throw new Error(gl.getProgramInfoLog(p));
    p.uniforms = {};
    return p;
  }

  circuitPieces(s, p = this.program) {
    const M = this.models, d = s.defense;
    for (let q of WORLD.doors)
      if (!d.doors[q.id])
        this.draw(M.phaseGate, mm(model(q.x, q.y, q.z, 0, q.axis === 'x' ? Math.PI / 2 : 0, 0), ms(q.width / 4.65, 1, 1)), p);
    for (let b of WORLD.breaches) {
      const angle = b.axis === 'x' ? Math.PI / 2 : 0;
      for (let i = 0; i < d.boards[b.id]; i++)
        this.draw(M.board, model(b.x, 0.95 + i * 0.3, b.z, 0, angle, i % 2 ? 0.035 : -0.035), p);
    }
    this.draw(M.centrifugeArm, model(0, 0, 2, 0, centrifugeAngle(s.time), 0), p);
    const elapsed = d.rocketLaunched ? s.time - d.rocketAt : 0;
    if (!d.rocketLaunched || elapsed < 25) {
      const lift = d.rocketLaunched ? Math.max(0, elapsed - 1) ** 2 * 1.25 : 0;
      this.draw(M.rocket, mt(56, lift + 0.2, 28), p);
      if (d.rocketLaunched)
        this.draw(M.rocketFire, mm(mt(56, lift + 0.2, 28), ms(1, 0.55 + Math.sin(elapsed * 30) * 0.12, 1)), p);
    }
    if (d.travel)
      this.draw(M.lander, mt(s.p.x, s.p.y - 0.08, s.p.z), p);
    if (d.power)
      for (let m of WORLD.machines)
        if (m.kind === 'perk' || m.kind === 'power')
          this.draw(M.indicator, mt(m.x, m.baseY + 2.11, m.z - 0.46), p);
    if (p !== this.program)
      return;
    for (let q of s.grenades)
      this.draw(M.grenade, model(q.x, q.y, q.z, s.time * 4, s.time * 3, 0), p);
    for (let q of d.pickups)
      this.draw(q.type === 'ammo' ? M.dropAmmo : q.type === 'double' ? M.dropDouble : q.type === 'insta' ? M.dropInsta : M.dropBonus, model(q.x, q.y + 0.15 + Math.sin(s.time * 3) * 0.1, q.z, 0, s.time, 0), p);
    for (let b of s.beams) {
      const age = s.time - b.at;
      if (age > (b.life || 0.25))
        continue;
      if (b.kind === 'explosion') {
        const k = 1 + age * 8;
        this.draw(M.gbeImpact, mm(mt(...b.end), ms(k, k, k)), p);
        continue;
      }
      const rays = b.rays?.length ? b.rays : [{
          start: b.start,
          end: b.end
        }];
      for (let q of rays) {
        const v = sub(q.end, q.start), L = Math.max(0.01, len(v)), basis = model(...q.start, -Math.asin(clamp(v[1] / L, -1, 1)), Math.atan2(v[0], v[2]), 0);
        this.draw(b.weapon === 'gbe' ? M.gbeBeamOuter : M.tracerMorrow, mm(basis, ms(1, 1, L)), p);
        if (b.weapon === 'gbe')
          this.draw(M.gbeBeamCore, mm(basis, ms(1, 1, L)), p);
      }
    }
  }
  shadowPass(buffer, vp, s) {
    let gl = this.gl, p = this.shadowProgram;
    gl.bindFramebuffer(gl.FRAMEBUFFER, buffer.fbo);
    gl.viewport(0, 0, buffer.size, buffer.size);
    gl.clear(gl.DEPTH_BUFFER_BIT);
    gl.enable(gl.DEPTH_TEST);
    gl.useProgram(p);
    this.m(p, 'uVP', vp);
    gl.enable(gl.POLYGON_OFFSET_FILL);
    gl.polygonOffset(1.6, 2);
    this.f(p, 'uTime', this.renderTime || s.time);
    this.drawStatic(vp, p);
    for (let e of s.enemies)
      this.zombieParts(e, p);
    this.circuitPieces(s, p);
    gl.disable(gl.POLYGON_OFFSET_FILL);
  }
  render(s) {
    if (this.frameFence) {
      let ready = this.gl.clientWaitSync(this.frameFence, 0, 0);
      if (ready === this.gl.TIMEOUT_EXPIRED)
        return false;
      this.gl.deleteSync(this.frameFence);
      this.frameFence = null;
    }
    this.resize();
    const gl = this.gl, p = this.program, pl = s.p, time = s.time + (menuOpen ? performance.now() / 1000 : 0);
    this.renderTime = time;
    this.updateSupplyPanel(s);
    let sway = Math.sin(time * 1.48) * 0.003 * settings.motion, shake = clamp(pl.shake || 0, 0, 1), baseEye = eyeOf(pl), eye = [
        baseEye[0] + Math.sin(time * 83) * shake * 0.012,
        baseEye[1] + Math.cos(time * 71) * shake * 0.018,
        baseEye[2] + Math.sin(time * 97) * shake * 0.012
      ], camPitch = pl.pitch + pl.recoil + sway + pl.hurt * 0.012 + Math.sin(time * 79) * shake * 0.012, camYaw = pl.yaw + Math.sin(time * 0.82) * 0.0016 * settings.motion + Math.sin(time * 67) * shake * 0.009, camRoll = (pl.roll || 0) + Math.sin(time * 61) * shake * 0.022, dir = [
        Math.sin(camYaw) * Math.cos(camPitch),
        Math.sin(camPitch),
        Math.cos(camYaw) * Math.cos(camPitch)
      ], up = [
        Math.sin(camRoll),
        Math.cos(camRoll),
        0
      ], view = lookAt(eye, add(eye, dir), up), character = settings.imaging === 'bodycam' ? 2 : settings.imaging === 'recorded' ? 1 : 0, sprintResponse = clamp(pl.sprintFov || 0, 0, 1), speedResponse = clamp(Math.hypot(pl.vx || 0, pl.vz || 0) / 5.5, 0, 1), fov = 88 + (character === 2 ? 19 : character === 1 ? 5 : 0) * settings.imageStrength + sprintResponse * 8 + (pl.sliding ? 1.5 : 0) + (pl.adrenaline > 0 ? 1.4 : 0) - 0, projection = perspective(mix(fov, 66, pl.aim) * Math.PI / 180, this.w / this.h, 0.035, 1600), vp = mm(projection, view), cameraZone = zoneAt(s);
    this.frameEye = eye;
    let ranked = WORLD.lights.map((l, i) => ({
        ...l,

        score: l.power / (1 + len(sub(l.p, eye)) ** 2)
      })).sort((a, b) => b.score - a.score), lamps = [
        ranked[0],
        ...ranked.slice(1).sort((a, b) => len(sub(a.p, eye)) - len(sub(b.p, eye))).slice(0, 11)

    while (lamps.length < 12)
      lamps.push({
        p: [
          0,

          0
        ],
        c: [
          0,
          0,
          0
        ],
        power: 0,
        radius: 1,
        source: -1
      });
    let key = lamps[0], keyId = key.source;
    if (this.keySource !== keyId || len(sub(this.keyFocus || eye, eye)) > 1.7) {
      this.keyVP = mm(perspective(2.3, 1, 0.12, 52), lookAt(key.p, eye, Math.abs(key.p[1] - eye[1]) < 0.3 ? [
        0,
        1,
        0
      ] : [
        0,
        0,
        -1
      ]));
      this.keySource = keyId;
      this.keyFocus = eye.slice();
      this.shadowDirty = true;
    }
    let flashPower = flashlightPowerFor(pl, false, time);
    let flashPos = add(eye, [
        0.03,
        -0.14,
        0.015
      ]), lightVP = mm(perspective(1.24, 1, 0.12, 48), lookAt(flashPos, add(flashPos, dir), up));
    if (flashPower > 0)
      this.shadowPass(this.shadow, lightVP, s);
    else if (this.shadowDirty)
      this.shadowPass(this.shadow, lightVP, s);
    if (this.shadowDirty || s.tick % 18 === 0 && this.keyTick !== s.tick) {
      this.shadowPass(this.keyShadow, this.keyVP, s);
      this.keyTick = s.tick;
    }
    this.shadowDirty = false;
    gl.bindFramebuffer(gl.FRAMEBUFFER, this.scene.fbo);
    gl.viewport(0, 0, this.w, this.h);
    gl.clearColor(0.011, 0.015, 0.019, 1);
    gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);
    gl.useProgram(p);

  constructor() {

  start() {
    try {
      if (this.ctx) {
        this.ctx.resume();
        this.active = true;
        return;
      }
      const AC = window.AudioContext || window.webkitAudioContext;
      if (!AC)
        return;
      const c = this.ctx = new AC(), master = this.master = c.createGain();
      master.gain.value = settings.volume * 0.7;
      let limiter = c.createDynamicsCompressor();
      limiter.threshold.value = -12;
      limiter.knee.value = 1;
      limiter.ratio.value = 16;
      limiter.attack.value = 0.004;
      limiter.release.value = 0.16;
      master.connect(limiter);
      limiter.connect(c.destination);
      this.compressor = limiter;
      this.rooms = {};
      for (let [name, duration, level] of [
          [
            'short',
            1.08,
            0.19
          ],
          [
            'long',
            3.15,
            0.27
          ]
        ]) {
        let rev = c.createConvolver(), imp = c.createBuffer(2, Math.floor(c.sampleRate * duration), c.sampleRate);
        for (let ch = 0; ch < 2; ch++) {
          let a = imp.getChannelData(ch);
          for (let i = 0; i < a.length; i++) {
            let t = i / a.length;
            a[i] = (this.rng() * 2 - 1) * Math.pow(1 - t, name === 'long' ? 3.4 : 4.4) * (i > c.sampleRate * (name === 'long' ? 0.041 : 0.014) ? 1 : 0.1);
          }
        }
        rev.buffer = imp;
        let gain = c.createGain();
        gain.gain.value = level;
        rev.connect(gain);
        gain.connect(master);
        this.rooms[name] = {
          input: rev,
          gain
        };
      }
      this.wet = this.rooms.long.gain;
      this.noise = c.createBuffer(1, c.sampleRate, c.sampleRate);
      let a = this.noise.getChannelData(0);
      for (let i = 0; i < a.length; i++)
        a[i] = this.rng() * 2 - 1;
      let wind = c.createBufferSource();
      wind.buffer = this.noise;
      wind.loop = true;
      let lp = c.createBiquadFilter();
      lp.type = 'lowpass';
      lp.frequency.value = 165;
      this.wind = c.createGain();
      this.wind.gain.value = 0.027;
      wind.connect(lp);
      lp.connect(this.wind);
      this.wind.connect(master);
      wind.start();
      for (let f of [
          46.2,
          73.6,
          119.8
        ]) {
        let o = c.createOscillator(), g = c.createGain();
        o.type = 'sine';
        o.frequency.value = f;
        g.gain.value = f < 60 ? 0.014 : 0.006;
        o.connect(g);
        g.connect(master);
        this.beds.push({
          node: g,
          base: g.gain.value

      this.active = true;

    } catch (e) {
      console.warn('Audio remains optional:', e.message);

  reserve() {
    if (this.transients.size < this.limit)
      return;
    let oldest = this.transients.values().next().value;
    try {
      oldest.source.stop();
    } catch {
    }
    oldest.clean();
  }
  own(source, nodes, until) {
    let id = ++this.serial, clean = () => {
        if (!this.transients.has(id))
          return;
        for (let n of nodes) {
          try {
            n.disconnect();
          } catch {
          }
        }
        source.onended = null;
        this.transients.delete(id);
      };
    this.transients.set(id, {
      source,
      nodes,
      until,
      clean
    });
    source.onended = clean;
  }
  burst(duration, gain, freq, type = 'lowpass', pos = null, delay = 0, send = 1) {
    if (!this.ctx || !this.active)
      return;
    this.reserve();
    let c = this.ctx, now = c.currentTime + delay, b = c.createBufferSource();
    b.buffer = this.noise;
    b.playbackRate.value = 0.82 + this.rng() * 0.27;
    let f = c.createBiquadFilter();
    f.type = type;
    f.frequency.value = freq;
    f.Q.value = 0.83;
    let g = c.createGain();
    g.gain.setValueAtTime(0.0001, now);
    g.gain.linearRampToValueAtTime(gain, now + 0.004);
    g.gain.exponentialRampToValueAtTime(0.0001, now + duration);
    let route = this.route(pos, send);
    b.connect(f);
    f.connect(g);
    g.connect(route.input);
    this.own(b, [
      b,
      f,
      g,
      ...route.nodes
    ], now + duration + 0.03);
    b.start(now, this.rng() * 0.18);
    b.stop(now + duration + 0.03);
  }

function applyLook(dx, dy) {

function keysToInput() {
  input.forward = (heldKeys.has('KeyW') || heldKeys.has('ArrowUp') ? 1 : 0) - (heldKeys.has('KeyS') || heldKeys.has('ArrowDown') ? 1 : 0);
  input.strafe = (heldKeys.has('KeyD') || heldKeys.has('ArrowRight') ? 1 : 0) - (heldKeys.has('KeyA') || heldKeys.has('ArrowLeft') ? 1 : 0);
  input.sprint = heldKeys.has('ShiftLeft') || heldKeys.has('ShiftRight');
  input.crouch = crouchToggle || heldKeys.has('ControlLeft') || heldKeys.has('ControlRight');
}
function keyAction(code) {
  if (WEAPON_KEYS[code]) {
    const r = equipWeapon(WEAPON_KEYS[code]);
    if (!r.ok)
      say(S, r.error, 2);
    return;
  }
  if (code === 'KeyQ')
    cycleWeapon();
  if (code === 'KeyE')
    useInteract();
  if (code === 'KeyR')
    reload(S);
  if (code === 'KeyH')
    bandage(S);
  if (code === 'KeyB')
    replaceCell(S);
  if (code === 'KeyG')
    throwGrenade(S);
  if (code === 'KeyV')
    melee(S);
  if (code === 'KeyF') {
    if (nearWindow(S))
      repairBarrier(S);
    else
      lightSwitch();
  }
  if (code === 'Space' && !S.defense.travel)
    jump();
  if (code === 'KeyC') {
    if (input.sprint && (input.forward || input.strafe)) {
      input.slide = true;
      return;
    }
    crouchToggle = !crouchToggle;
    keysToInput();
  }
}

function frame(now) {
  requestAnimationFrame(frame);
  if (!S || !renderer)
    return;
  const elapsed = lastFrame ? Math.min((now - lastFrame) / 1000, 0.1) : STEP;
  lastFrame = now;
  if (run && !paused && !menuOpen && S.phase === 'alive') {
    accum += elapsed;
    while (accum >= STEP) {
      tick(S, STEP, input);
      input.slide = false;
      accum -= STEP;
      if (S.phase === 'dead') {
        showDefeat();
        break;
      }
    }
  } else
    accum = 0;
  try {
    const drawn = renderer.render(S);
    if (drawn) {
      const t = performance.now();
      renderer.fps = mix(renderer.fps, 1 / Math.max((t - (renderer.presentAt || t - 16.7)) / 1000, 0.001), 0.05);
      renderer.presentAt = t;
    }
    if (run && !paused)
      audio?.update(S);
    if (now - lastUi > 80) {
      updateHUD();
      lastUi = now;
    }
  } catch (e) {
    console.error(e);
    run = false;
    $('error').classList.remove('hidden');
    $('error-detail').textContent = e.message;
  }
  if (now > 8000 && renderer.fps < 30 && renderer.scale > 0.64) {
    renderer.scale = Math.max(0.64, renderer.scale - 0.12);
    renderer.fps = 50;
  }
}

function walkableSegment(a, b) {
  for (let i = 1; i <= 8; i++) {
    const t = i / 8, x = mix(a.x, b.x, t), z = mix(a.z, b.z, t), y = mix(a.y || 0, b.y || 0, t), f = floorAt(x, z, y);
    if (!f || Math.abs(f.y - y) > 0.36 || blocked(x, f.y, z, 0.28, 1.8))
      return false;
  }
  return true;
}
function nearWindow(s) {
  return WORLD.breaches.filter(b => s.defense.rooms[b.room] && s.defense.boards[b.id] < 6 && Math.hypot(s.p.x - b.inside.x, s.p.z - b.inside.z) < 2.4 && Math.abs(s.p.y - (floorAt(b.inside.x, b.inside.z)?.y || 0)) < 0.5).sort((a, b) => dist2(a.inside, s.p) - dist2(b.inside, s.p))[0];
}
function repairBarrier(s) {
  const b = nearWindow(s);
  if (!b)
    return {
      ok: false,
      error: 'Stand by a damaged window.'
    };
  if (s.time < s.defense.repairClock)
    return { ok: false };
  s.defense.repairClock = s.time + 0.45;
  s.defense.boards[b.id]++;
  s.defense.trace += 10;
  s.pointFx.push({
    n: 10,
    at: s.time
  });
  audio?.boardRepair([
    b.x,
    1.2,
    b.z
  ]);
  return {
    ok: true,
    barrier: b.id
  };
}
function dirtBurst(s, p, n = 8) {
  for (let i = 0; i < n; i++)
    s.particles.push({
      x: p[0] + (s.rng() - 0.5) * 0.3,
      y: p[1],
      z: p[2] + (s.rng() - 0.5) * 0.3,
      vx: (s.rng() - 0.5) * 3,
      vy: 0.6 + s.rng() * 2,
      vz: (s.rng() - 0.5) * 3,
      life: 0.25 + s.rng() * 0.4,
      mat: 15
    });
}
function throwGrenade(s) {
  const p = s.p;
  if (s.phase !== 'alive' || p.grenadeCount <= 0 || p.grenadeCd > 0 || s.defense.travel)
    return { ok: false };
  p.grenadeCount--;
  p.grenadeCd = 0.65;
  const f = forward(p), o = eyeOf(p);
  s.grenades.push({
    x: o[0] + f[0] * 0.3,
    y: o[1] - 0.1,
    z: o[2] + f[2] * 0.3,
    vx: f[0] * 10,
    vy: f[1] * 10 + 3.4,
    vz: f[2] * 10,
    life: 1.55
  });
  audio?.click();
  return { ok: true };
}
function explodeGrenade(s, q) {
  const p = [
    q.x,
    q.y + 0.15,
    q.z
  ];
  dirtBurst(s, p, 35);
  audio?.explosion(p);
  s.beams.push({
    id: 'explosion' + s.time,
    kind: 'explosion',
    weapon: 'grenade',
    start: p,
    end: p,
    at: s.time,
    life: 0.35
  });
  for (let e of s.enemies) {
    let d = Math.hypot(e.x - q.x, e.z - q.z, e.y + 1 - q.y);
    if (e.hp <= 0 || d > 5.3 || !canSee(p, [
        e.x,
        e.y + 1.2,
        e.z
      ]))
      continue;
    const damage = Math.max(20, 330 * (1 - d / 6));
    e.hp = Math.max(0, e.hp - damage);
    e.stagger = 1.2;
    e.knockX += (e.x - q.x) * 0.8;
    e.knockZ += (e.z - q.z) * 0.8;
    if (e.hp === 0) {
      e.deadAt = s.time;
      e.state = 'inert';
      awardTrace(12, 'kill');
    } else
      awardTrace(12, 'contact');
  }
  if (!s.defense.travel && Math.hypot(s.p.x - q.x, s.p.z - q.z, s.p.y + 1 - q.y) < 3.8 && canSee(p, eyeOf(s.p)))
    hurt(s, Math.max(12, 70 * (1 - Math.hypot(s.p.x - q.x, s.p.z - q.z) / 4)), 'grenade');
}
function melee(s) {
  const p = s.p;
  if (p.meleeCd > 0 || s.phase !== 'alive' || s.defense.travel)
    return { ok: false };
  p.meleeCd = 0.7;
  p.shake = 0.15;
  let target = s.enemies.filter(e => e.hp > 0 && dist2(e, p) < 1.9 && Math.abs(e.y - p.y) < 1 && dot(norm([
    e.x - p.x,
    0,
    e.z - p.z
  ]), [
    Math.sin(p.yaw),
    0,
    Math.cos(p.yaw)
  ]) > 0.35 && canSee(eyeOf(p), [
    e.x,
    e.y + 1.4,
    e.z
  ])).sort((a, b) => dist2(a, p) - dist2(b, p))[0];
  audio?.zombieStrike(eyeOf(p));
  if (!target)
    return {
      ok: true,
      hit: false
    };
  target.hp = Math.max(0, target.hp - 150 * (1 + (s.defense.weaponTiers[p.weapon] || 0)));
  target.stagger = 0.8;
  if (target.hp === 0) {
    target.deadAt = s.time;
    target.state = 'inert';
    awardTrace(0, 'melee');
  } else
    awardTrace(12, 'contact');
  s.feedback = {
    at: s.time,
    hit: true,
    critical: false,
    damage: 150
  };
  return {
    ok: true,
    hit: true
  };
}
function spawnDrop(s) {
  const e = s.enemies.filter(e => e.hp === 0).sort((a, b) => b.deadAt - a.deadAt)[0];
  if (!e)
    return;
  const type = s.defense.dropCount++ % 2 ? 'bonus' : 'ammo';
  s.defense.pickups.push({
    type,
    x: e.x,
    y: e.y + 0.3,
    z: e.z,
    at: s.time,
    life: 18
  });
}
function defenseStep(s, dt) {
  const p = s.p, d = s.defense;
  p.grenadeCd = Math.max(0, p.grenadeCd - dt);
  p.meleeCd = Math.max(0, p.meleeCd - dt);
  p.bleed = p.injury = 0;
  if (s.time - p.lastHurt > 5 && p.hp < p.maxHp)
    p.hp = Math.min(p.maxHp, p.hp + dt * p.maxHp * 0.24);
  if (d.power)
    p.battery = Math.max(p.battery, 119);
  if (heldKeys.has('KeyF') && nearWindow(s))
    repairBarrier(s);
  if (d.travel) {
    const q = d.travel;
    q.t = Math.min(q.duration, q.t + dt);
    const u = q.t / q.duration, t = u * u * (3 - 2 * u);
    p.x = mix(q.from.x, q.to.x, t);
    p.z = mix(q.from.z, q.to.z, t);
    p.y = mix(q.from.y, q.to.y, t) + Math.sin(u * Math.PI) * 8;
    p.ground = false;
    p.vx = p.vz = p.vy = 0;
    if (u >= 1) {
      p.y = floorAt(q.to.x, q.to.z)?.y || 0;
      p.ground = true;
      if (q.pad !== 'home')
        d.landerUsed[q.pad] = true;
      d.travel = null;
      say(s, 'LANDER ARRIVED', 2);
    }
  }
  const spin = s.time % 32;
  if (!d.travel && spin > 21 && spin < 27 && s.time - d.centrifugeHitAt > 1.4 && p.y < 1.15) {
    let dx = p.x, dz = p.z - 2, a = centrifugeAngle(s.time), side = Math.abs(dx * Math.cos(a) - dz * Math.sin(a)), radius = Math.hypot(dx, dz);
    if (radius > 1.45 && radius < 4.65 && side < 0.45) {
      d.centrifugeHitAt = s.time;
      hurt(s, 35, 'centrifuge');
      let L = radius || 1;
      p.vx += dx / L * 7;
      p.vz += dz / L * 7;
      say(s, 'KEEP CLEAR OF THE CENTRIFUGE', 2);
    }
  }
  if (d.rocketLaunched && !d.doors.rocketDoor && s.time - d.rocketAt >= 6) {
    d.doors.rocketDoor = true;
    d.rooms.launch = true;
    renderer && (renderer.shadowDirty = true);
    say(s, 'ARSENAL OPEN \u2014 UPGRADE WEAPONS AT THE LAUNCH COMPLEX', 5);
  }
  for (let q of s.grenades) {
    q.life -= dt;
    const nx = q.x + q.vx * dt, nz = q.z + q.vz * dt;
    q.vy -= dt * 11;
    if (blocked(nx, q.y, nz, 0.07, 0.1)) {
      q.vx *= -0.45;
      q.vz *= -0.45;
    } else {
      q.x = nx;
      q.z = nz;
    }
    q.y += q.vy * dt;
    let f = floorAt(q.x, q.z);
    if (f && q.y < f.y + 0.09) {
      q.y = f.y + 0.09;
      q.vy = Math.abs(q.vy) * 0.3;
      q.vx *= 0.65;
      q.vz *= 0.65;
    }
    if (q.life <= 0)
      explodeGrenade(s, q);
  }
  s.grenades = s.grenades.filter(q => q.life > 0);
  for (let q of d.pickups) {
    q.life -= dt;
    if (q.life > 0 && dist2(q, p) < 1.1 && Math.abs(p.y - q.y) < 1.4) {
      q.life = 0;
      if (q.type === 'ammo') {
        recordAmmo(p);
        for (let id of Object.keys(p.ownedWeapons))
          p.ownedWeapons[id].reserve = weaponSpec(id).initialReserve;
        p.reserve = weaponSpec(p.weapon).initialReserve;
        p.grenadeCount = 4;
        audio?.perkCue();
        say(s, 'MAX AMMO', 3);
      } else if (q.type === 'double') {
        d.doubleUntil = s.time + 30;
        audio?.perkCue();
        say(s, 'DOUBLE POINTS', 3);
      } else if (q.type === 'insta') {
        d.instaUntil = s.time + 20;
        audio?.perkCue();
        say(s, 'INSTA-KILL', 3);
      } else {
        d.trace += 500;
        s.pointFx.push({
          n: 500,
          at: s.time
        });
        say(s, 'BONUS POINTS +500', 3);
        audio?.perkCue();
      }
    }
  }
  d.pickups = d.pickups.filter(q => q.life > 0);
  s.pointFx = s.pointFx.filter(q => s.time - q.at < 1.5);
}
function prepareRound(s) {
  const d = s.defense;
  d.monkeyRound = d.wave % 5 === 0 && d.power && Object.values(d.perks).some(Boolean);
  d.perkDamage = {
    jug: 0,
    quick: 0,
    stamina: 0
  };
  d.queue = d.monkeyRound ? Math.min(10, 3 + Math.floor(d.wave / 5)) : waveCount(d.wave);
  d.spawned = 0;
  d.spawnClock = d.monkeyRound ? 2 : 2.3;
}
function spawnMonkey(s) {
  const d = s.defense, idx = ++d.spawned, targets = WORLD.machines.filter(m => m.kind === 'perk' && d.perks[m.perk]);
  const m = targets[(idx - 1) % Math.max(1, targets.length)], pad = WORLD.pads.find(p => p.room === m?.room) || WORLD.pads[0];
  let x = pad.x + (idx % 2 ? 0.7 : -0.7), z = pad.z - 1.2, f = floorAt(x, z);
  if (!f || blocked(x, f.y, z, 0.23, 0.95)) {
    x = pad.x;
    z = pad.z;
    f = floorAt(x, z);
  }
  const ground = f?.y || 0, hp = 140 + Math.max(0, d.wave - 5) * 25;
  s.enemies.push({
    id: 'M' + d.wave + '-' + idx,
    archetype: 'space-monkey',
    variant: idx % 3,
    x,
    y: ground + 5,
    z,
    hp,
    maxHp: hp,
    headHeight: 0.79,
    bodyHeight: 0.43,
    headRadius: 0.23,
    bodyRadius: 0.3,
    state: 'drop',
    dropGround: ground,
    yaw: 0,
    target: null,
    targetMachine: m?.id || null,
    path: [],
    pathTimer: 0,
    alert: 35,
    attack: 0,
    attackResolved: false,
    attackDamage: 34,
    anim: s.rng() * 6,
    step: 0,
    deadAt: 0,
    stagger: 0,
    knockX: 0,
    knockZ: 0
  });
  audio?.monkeyCue([
    x,
    ground + 1,
    z
  ]);
  return true;
}
function monkeyTick(s, e, dt) {
  if (e.hp <= 0)
    return;
  const d = s.defense, p = s.p;
  e.stagger = Math.max(0, e.stagger - dt);
  e.attack = Math.max(0, e.attack - dt);
  e.knockX *= Math.exp(-dt * 7);
  e.knockZ *= Math.exp(-dt * 7);
  e.anim += dt * 8;
  if (e.state === 'drop') {
    e.y = Math.max(e.dropGround, e.y - dt * 7);
    if (e.y === e.dropGround) {
      e.state = 'hunt';
      dirtBurst(s, [
        e.x,
        e.y + 0.1,
        e.z
      ], 12);
    }
    return;
  }
  let machine = WORLD.machines.find(m => m.id === e.targetMachine && d.perks[m.perk]);
  if (!machine) {
    machine = WORLD.machines.filter(m => m.kind === 'perk' && d.perks[m.perk]).sort((a, b) => dist2(a, e) - dist2(b, e))[0];
    e.targetMachine = machine?.id || null;
  }
  const target = machine ? {
    x: machine.x,
    y: machine.baseY,
    z: machine.z - 1.05
  } : {
    x: p.x,
    y: p.y,
    z: p.z
  };
  e.target = target;
  const distance = dist2(e, target);
  if (machine && distance < 0.92 && Math.abs(e.y - target.y) < 0.6) {
    if (e.attack === 0) {
      e.attack = 0.85;
      audio?.monkeyCue([
        e.x,
        e.y + 0.7,
        e.z
      ]);
    }
    if (e.stagger === 0)
      d.perkDamage[machine.perk] = (d.perkDamage[machine.perk] || 0) + dt * 32;
    if (d.perkDamage[machine.perk] >= 100) {
      d.perks[machine.perk] = false;
      e.targetMachine = null;
      if (machine.perk === 'jug') {
        p.maxHp = 100;
        p.hp = Math.min(p.hp, 100);
      }
      say(s, machine.name + ' STOLEN \u2014 DEFEND THE REMAINING MACHINES', 4);
      audio?.impact([
        machine.x,
        machine.y,
        machine.z
      ]);
    }
    return;
  }
  if (!machine && distance < 1.5 && Math.abs(p.y - e.y) < 1.1 && !d.travel && e.attack === 0 && canSee([
      e.x,
      e.y + 0.8,
      e.z
    ], eyeOf(p))) {
    e.attack = 1.05;
    hurt(s, 34, 'space monkeys');
    audio?.monkeyCue([
      e.x,
      e.y + 0.7,
      e.z
    ]);
  }
  e.pathTimer -= dt;
  const direct = walkableSegment(e, target) && canSee([
    e.x,
    e.y + 0.7,
    e.z
  ], [
    target.x,
    target.y + 0.7,
    target.z
  ]);
  if (e.pathTimer <= 0) {
    e.path = direct ? [target] : pathBetween(e, target);
    e.pathTimer = 0.5;
  }
  while (e.path.length > 1 && dist2(e, e.path[0]) < 0.42)
    e.path.shift();
  const goal = e.path[0];
  if (!goal)
    return;
  const dx = goal.x - e.x, dz = goal.z - e.z, L = Math.hypot(dx, dz) || 1, speed = e.stagger > 0 ? 0.45 : 3.45, nx = e.x + dx / L * speed * dt + e.knockX * dt, nz = e.z + dz / L * speed * dt + e.knockZ * dt;
  e.yaw += angleDelta(Math.atan2(dx, dz), e.yaw) * Math.min(1, dt * 8);
  let f = floorAt(nx, nz, e.y);
  if (f && !blocked(nx, f.y, nz, 0.23, 0.98)) {
    e.x = nx;
    e.z = nz;
    e.y = f.y;
  } else {
    let a = floorAt(nx, e.z, e.y);
    if (a && !blocked(nx, a.y, e.z, 0.23, 0.98)) {
      e.x = nx;
      e.y = a.y;
    } else {
      a = floorAt(e.x, nz, e.y);
      if (a && !blocked(e.x, a.y, nz, 0.23, 0.98)) {
        e.z = nz;
        e.y = a.y;
      } else
        e.pathTimer = 0;
    }
  }
}
function spawnDrop(s) {
  const e = s.enemies.filter(e => e.hp === 0).sort((a, b) => b.deadAt - a.deadAt)[0];
  if (!e)
    return;
  const types = [
      'ammo',
      'bonus',
      'double',
      'insta'
    ], type = types[s.defense.dropCount++ % types.length];
  s.defense.pickups.push({
    type,
    x: e.x,
    y: e.y + 0.3,
    z: e.z,
    at: s.time,
    life: 18
  });
}
function centrifugeAngle(time) {
  return (Math.floor(time / 32) * 6 + clamp(time % 32 - 21, 0, 6)) * 1.35;
}
